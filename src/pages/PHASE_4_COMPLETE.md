### Using Bulk Operations in Tables

```typescript
import { BulkOperationToolbar, commonBulkOperations } from '@/components/features/BulkOperationToolbar'

const [selectedIds, setSelectedIds] = useState<string[]>([])

const operations = [
  commonBulkOperations.markCompliant(async (ids) => {
    await supabase.from('vehicles').update({ compliant: true }).in('id', ids)
  }),
  commonBulkOperations.delete(async (ids) => {
    await supabase.from('observations').delete().in('id', ids)
  }),
]

// In render:
<BulkOperationToolbar
  selectedIds={selectedIds}
  totalItems={items.length}
  onSelectAll={() => setSelectedIds(items.map(i => i.id))}
  onClearSelection={() => setSelectedIds([])}
  operations={operations}
  itemName="vehicles"
/>
```
